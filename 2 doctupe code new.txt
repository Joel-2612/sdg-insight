                            // Find the country with the highest progress for this SDG
                            const topCountry = sdg.countries.reduce((prev, current) => (prev.progress > current.progress) ? prev : current);
                            return `
                                <div class="p-6 rounded-lg shadow-md ${sdg.color.replace('sdg-color', 'bg')} bg-opacity-10 text-gray-800">
                                    <h3 class="text-2xl font-bold mb-2 ${sdg.color.replace('sdg-color', 'sdg-text-color')}">${sdg.id}. ${sdg.title}</h3>
                                    <p class="text-md mb-3">${sdg.description.substring(0, 150)}...</p>
                                    <p class="text-lg font-semibold text-gray-700 mb-4">Top Country for Effective Implementation: <span class="sdg-text-color-${sdg.id}">${topCountry.name}</span></p>
                                    <div class="image-display-container">
                                        <img src="${sdg.imageUrl}" alt="Image related to SDG ${sdg.id} in ${topCountry.name}">
                                    </div>
                                </div>
                            `;
                        }).join('')}
                    </div>
                `;
            }
            else if (sdgId === 'sources') {
                contentHTML = `
                    <h2 class="text-3xl font-bold mb-6 text-gray-800">📚 Sources for Sustainable Development Goals</h2>
                    <div class="space-y-6">
                        ${sdgData.map(sdg => `
                            <div>
                                <h3 class="text-xl font-semibold text-gray-800">${sdg.id}: ${sdg.title}</h3>
                                <p class="text-gray-700">${sdg.description}</p>
                            </div>
                        `).join('')}
                    </div>
                `;
            } else {
                const sdg = sdgData.find(s => s.id === sdgId);
                if (sdg) {
                    contentHTML = `
                        <div class="sdg-title-banner ${sdg.color}">
                            ${sdg.id}. ${sdg.title}
                        </div>
                        <div class="mb-6 p-4 bg-gray-100 rounded-lg shadow-inner">
                            <h3 class="text-xl font-semibold mb-2 text-gray-800">Worldwide Progress:</h3>
                            <div class="progress-container">
                                <div class="progress-bar ${sdg.color}" style="width: ${sdg.progress.worldwide}%;">
                                    ${sdg.progress.worldwide}%
                                </div>
                            </div>
                        </div>

                        <p class="text-lg leading-relaxed mb-6 text-gray-700">${sdg.description}</p>

                        <h3 class="text-2xl font-semibold mb-3 text-gray-800">Top Countries for Effective Implementation:</h3>
                        <div class="grid grid-cols-1 lg:grid-cols-2 gap-4 mb-6">
                            ${sdg.countries.map(country => `
                                <div class="country-card ${sdg.countryDetails && sdg.countryDetails[country.name] ? 'cursor-pointer hover:bg-gray-700' : ''}"
                                     ${sdg.countryDetails && sdg.countryDetails[country.name] ? `onclick="showCountryDetailsModal(${sdg.id}, '${country.name}')"` : ''}>
                                    <div class="country-info">
                                        <img src="${getCountryFlagUrl(country.name)}" alt="Flag of ${country.name}" class="country-flag mr-4">
                                        <span class="text-lg text-gray-800 font-medium">${country.name}</span>
                                    </div>
                                    <div class="progress-container">
                                        <div class="progress-bar ${sdg.color}" style="width: ${country.progress}%;">
                                            ${country.progress}% Implemented
                                        </div>
                                    </div>
                                </div>
                            `).join('')}
                        </div>
                        
                        <h3 class="text-2xl font-semibold mb-3 text-gray-800">Key Statistics:</h3>
                        <div class="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4 mb-8">
                            ${sdg.statistics.map(stat => `
                                <div class="stat-box">
                                    <div class="stat-number">${stat.number}</div>
                                    <div class="stat-label">${stat.label}</div>
                                </div>
                            `).join('')}
                        </div>

                        <h3 class="text-2xl font-semibold mb-3 text-gray-800">Key Initiatives/Projects:</h3>
                        <ul class="list-disc list-inside text-gray-700 space-y-2 mb-6">
                            ${sdg.initiatives.map(initiative => `
                                <li class="text-lg">${initiative}</li>
                            `).join('')}
                        </ul>
                    `;
                } else {
                    contentHTML = '<p class="text-red-500">SDG not found.</p>';
                }
            }
            contentDisplay.innerHTML = contentHTML;
            window.scrollTo({ top: 0, behavior: 'smooth' });

            if (sdgId === 'home') {
                document.querySelectorAll('[data-sdg-id]').forEach(card => {
                    card.addEventListener('click', (event) => {
                        const clickedSdgId = parseInt(event.currentTarget.dataset.sdgId);
                        displaySDG(clickedSdgId);
                        document.querySelectorAll('#sdg-nav a').forEach(link => {
                            link.classList.remove('active');
                        });
                        document.querySelector(`#sdg-nav a[data-sdg="${clickedSdgId}"]`).classList.add('active');
                    });
                });
            }
        }

        sdgData.forEach(sdg => {
            const li = document.createElement('li');
            li.className = 'mb-2';
            li.innerHTML = `
                <a href="#" data-sdg="${sdg.id}" class="block p-3 rounded-md hover:bg-gray-700 transition-colors duration-200 text-white flex items-center">
                    <span class="sdg-sidebar-number ${sdg.color}">${sdg.id}</span>
                    <span class="font-medium">${sdg.title}</span>
                </a>
            `;
            sdgNav.appendChild(li);
        });

        const aboutSdgLi = document.createElement('li');
        aboutSdgLi.className = 'mb-2';
        aboutSdgLi.innerHTML = `
            <a href="#" data-sdg="about-sdg" class="block p-3 rounded-md hover:bg-gray-700 transition-colors duration-200 text-white flex items-center">
                <span class="sdg-sidebar-number bg-green-500">📖</span>
                <span class="font-medium">About SDG</span>
            </a>
        `;
        sdgNav.insertBefore(aboutSdgLi, sdgNav.children[1]);

        const imagesLi = document.createElement('li');
        imagesLi.className = 'mb-2';
        imagesLi.innerHTML = `
            <a href="#" data-sdg="images" class="block p-3 rounded-md hover:bg-gray-700 transition-colors duration-200 text-white flex items-center">
                <span class="sdg-sidebar-number bg-red-500">🖼️</span>
                <span class="font-medium">Images</span>
            </a>
        `;
        sdgNav.appendChild(imagesLi);

        const sourcesLi = document.createElement('li');
        sourcesLi.className = 'mb-2';
        sourcesLi.innerHTML = `
            <a href="#" data-sdg="sources" class="block p-3 rounded-md hover:bg-gray-700 transition-colors duration-200 text-white flex items-center">
                <span class="sdg-sidebar-number bg-purple-500">📚</span>
                <span class="font-medium">Sources</span>
            </a>
        `;
        sdgNav.appendChild(sourcesLi);

        sdgNav.addEventListener('click', (event) => {
            event.preventDefault();
            const target = event.target.closest('a');
            if (target) {
                const sdgId = target.dataset.sdg;
                if (sdgId) {
                    displaySDG(sdgId === 'home' ? 'home' : (sdgId === 'sources' ? 'sources' : (sdgId === 'about-sdg' ? 'about-sdg' : (sdgId === 'images' ? 'images' : parseInt(sdgId)))));
                    
                    document.querySelectorAll('#sdg-nav a').forEach(link => {
                        link.classList.remove('active');
                    });
                    target.classList.add('active');
                }
            }
        });

        document.addEventListener('DOMContentLoaded', () => {
            displaySDG('home');
            document.querySelector('#sdg-nav a[data-sdg="home"]').classList.add('active');
        });
    </script>
